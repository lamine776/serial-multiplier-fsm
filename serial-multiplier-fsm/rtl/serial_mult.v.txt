`timescale 1ns / 1ps

module serial_mult #(
    parameter WIDTH = 8
)(
    input  wire               clk,
    input  wire               rst,
    input  wire               start,
    input  wire [WIDTH-1:0]   A,
    input  wire [WIDTH-1:0]   B,
    output reg  [2*WIDTH-1:0] product,
    output reg                done
);

    // =========================================================
    // FSM state encoding
    // =========================================================
    localparam IDLE    = 2'b00;
    localparam CALC    = 2'b01;
    localparam DONE_ST = 2'b10;

    reg [1:0] state_reg, state_next;

    // =========================================================
    // Datapath registers
    // =========================================================
    reg [2*WIDTH-1:0]     acc;       // 16-bit accumulator
    reg [WIDTH-1:0]       a_reg;    // multiplicand (constant during operation)
    reg [WIDTH-1:0]       b_reg;    // multiplier shift register
    reg [$clog2(WIDTH):0] count;    // iteration counter

    // =========================================================
    // Status signals: datapath -> FSM
    // =========================================================
    wire b_lsb      = b_reg[0];
    wire count_done  = (count == WIDTH-1);

    // =========================================================
    // Block 1: State register (sequential)
    // =========================================================
    always @(posedge clk) begin
        if (rst)
            state_reg <= IDLE;
        else
            state_reg <= state_next;
    end

    // =========================================================
    // Block 2: Next-state logic (combinational)

    always @(*) begin
        state_next = state_reg;
        case (state_reg)
            IDLE: begin
                if (start)
                    state_next = CALC;
            end
            CALC: begin
                if (count_done == 1)
                    state_next = DONE_ST;
                else
                    state_next = CALC;
            end
            DONE_ST: begin
                if (rst)
                    state_next = IDLE;
                else
                    state_next = DONE_ST;
            end
            default: state_next = IDLE; 
        endcase
    end

    // =========================================================
    // Block 3: Datapath operations (sequential)
    // The FSM state determines what the datapath does each cycle.

    always @(posedge clk) begin
        if (rst) begin
            acc     <= 0;
            a_reg   <= 0;
            b_reg   <= 0;
            count   <= 0;
            product <= 0;
            done    <= 0;
        end else begin
            case (state_reg)
                IDLE: begin
                    done <= 0;
                    if (start) begin
                        // TODO: Load operands into registers.
                        //       Clear the accumulator and counter.
                        //       (4 lines: a_reg, b_reg, acc, count)
                        a_reg <= A;
                        b_reg <= B;
                        acc <= 0;
                        count <= 0;
                    end
                end

                CALC: begin
                    
                    if(b_lsb) begin
                        acc <= ( {1'b0, acc} + {1'b0, a_reg, {WIDTH{1'b0}}} ) >> 1;
                    end else begin
                        acc <= acc >> 1;
                    end
                    
                    b_reg <= b_reg >> 1;
                    count <= count + 1;
                end

                DONE_ST: begin
                    // TODO: Copy the accumulator to the product output.
                    //       Assert the done flag.
                    product <= acc;
                    done    <= 1;
                end
            endcase
        end
    end
endmodule
